用导出的 TensorRT engine 文件做推理¶

上一篇（0817） 我们用 ultralytics 导出了 3 份部署文件，其中两份是 TensorRT 引擎：

  • D:\project\step1\week12\yolov8n_fp16.engine — 半精度（体积减半）
  • D:\project\step1\week12\yolov8n_int8.engine — INT8 量化（体积 1/4，最快）

本节目标：直接把 .engine 文件拿来跑推理，验证"导出 → 加载 → 推理"全链路。

⚠️ 提醒：.engine 是绑定本机 GPU（RTX 3060, SM 8.6）和 TensorRT 10.10 的，换电脑需要重新导出。

import os
import tensorrt as trt
import matplotlib.pyplot as plt
from PIL import Image
from ultralytics import YOLO

import sys
sys.path.insert(0, r"d:\project\step1\week14")
from trt_utils import load_engine, print_engine_io

# ============================================================
# ① 反序列化 engine 文件，确认能加载并查看输入/输出
# ============================================================
eng_path = r"D:\project\step1\week12\yolov8n_fp16.engine"
print(f"引擎文件: {eng_path} ({os.path.getsize(eng_path) / 1024 / 1024:.1f} MB)")
print(f"TensorRT 版本: {trt.__version__}")

engine = load_engine(eng_path)          # 自动处理 ultralytics 元数据头
print("✓ 引擎反序列化成功\n")
print("引擎输入/输出张量:")
print_engine_io(engine)

# ============================================================
# ② 用导出的 engine 直接做推理（ultralytics 底层走 TensorRT）
# ============================================================
model = YOLO(eng_path)          # 直接加载 .engine 文件
print(f"\n模型任务: {model.task} | 类别数: {len(model.names)}")

# 用 COCO 测试图 bus.jpg（公交车 + 行人 + 交通灯，正好覆盖常用类别）
img = r"D:\project\step1\week12\ultralytics-8.4.113\ultralytics\assets\bus.jpg"
results = model.predict(img, conf=0.5, save=True,
                        project=r"D:\project\step1\week14", name="yolo_engine_out")

r = results[0]
names = model.names
print(f"\n检测到 {len(r.boxes)} 个目标:")
for box in r.boxes:
    cls, conf = int(box.cls), float(box.conf)
    xyxy = [round(v, 1) for v in box.xyxy[0].tolist()]
    print(f"  {names[cls]:12s} 置信度={conf:.3f}  框={xyxy}")

# 显示带框的推理结果图
saved_img = r"D:\project\step1\week14\yolo_engine_out\bus.jpg"
plt.figure(figsize=(10, 6))
plt.imshow(Image.open(saved_img))
plt.axis("off")
plt.title("TensorRT FP16 engine 推理结果 (yolov8n_fp16.engine)")
plt.show()

from ultralytics import YOLO
import time

# ============================================================
# 对比 FP16 和 INT8 两个引擎：速度 + 精度
# ============================================================
img = r"D:\project\step1\week12\ultralytics-8.4.113\ultralytics\assets\bus.jpg"
engines = {
    "FP16 (半精度)": r"D:\project\step1\week12\yolov8n_fp16.engine",
    "INT8 (量化)  ": r"D:\project\step1\week12\yolov8n_int8.engine",
}

for name, path in engines.items():
    model = YOLO(path)
    # 预热（引擎首次加载含 kernel 编译/显存分配，不计入统计）
    model.predict(img, conf=0.5, verbose=False)

    # 跑 30 次取平均耗时
    times = []
    for _ in range(30):
        s = time.time()
        r = model.predict(img, conf=0.5, verbose=False)[0]
        times.append(time.time() - s)

    avg = sum(times) / len(times) * 1000
    n_objs = len(r.boxes)
    top_conf = max((float(b.conf) for b in r.boxes), default=0)
    print(f"{name}  平均推理耗时 {avg:6.2f} ms/张 | 检测到 {n_objs} 个目标 | 最高置信度 {top_conf:.3f}")

print("\n对比结论：INT8 通常比 FP16 更快、体积更小，精度略有下降（本项目场景下几乎无感）")
