\usepackage{setspace}
\onehalfspacing

\setlength{\parskip}{0pt}

\usepackage{sectsty}
\allsectionsfont{\sffamily\bfseries}

% ---- 行内代码等宽字体中的中文正常显示 ----
\xeCJKsetup{CJKmath=true}

% ---- 避免图片/表格孤行 ----
